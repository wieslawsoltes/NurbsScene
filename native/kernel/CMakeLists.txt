cmake_minimum_required(VERSION 3.20)
project(NurbsSceneKernel VERSION 0.1.0 LANGUAGES C)
add_library(nurbsscene_kernel SHARED src/nurbsscene.c)
target_include_directories(nurbsscene_kernel PUBLIC include)
target_compile_features(nurbsscene_kernel PRIVATE c_std_11)
set_target_properties(nurbsscene_kernel PROPERTIES C_VISIBILITY_PRESET hidden)
if(NOT MSVC)
 target_compile_options(nurbsscene_kernel PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
 target_link_libraries(nurbsscene_kernel PRIVATE m)
endif()
install(TARGETS nurbsscene_kernel)
install(FILES include/nurbsscene.h DESTINATION include)
